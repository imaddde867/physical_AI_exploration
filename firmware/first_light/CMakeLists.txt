cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH})
    message(FATAL_ERROR "Set PICO_SDK_PATH to the Raspberry Pi pico-sdk checkout")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(buddy_first_light C CXX ASM)
pico_sdk_init()

add_executable(buddy_first_light main.c)
target_link_libraries(buddy_first_light pico_stdlib hardware_spi)
pico_enable_stdio_usb(buddy_first_light 1)
pico_enable_stdio_uart(buddy_first_light 0)
pico_add_extra_outputs(buddy_first_light)
